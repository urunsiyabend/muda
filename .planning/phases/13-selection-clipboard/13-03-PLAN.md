fn handle_copy(&mut self, ctx: &CommandContext) {
    if let Some((start, end)) = ctx.view.selection_range() {
        // Selection active: copy selected text (existing behavior)
        let text = ctx.document.slice(TextRange::new(start, end));
        if let Some(ref mut cb) = self.clipboard {
            let _ = cb.set_text(text);
        }
        self.clipboard_is_line_copy = false;
    } else {
        // No selection: copy entire current line including newline
        let pos = self.offset_to_position(ctx, ctx.view.caret_offset());
        let line_start = self.position_to_offset(ctx, TextPosition::new(pos.line, 0));
        let line_end = if pos.line + 1 < ctx.document.len_lines() {
            self.position_to_offset(ctx, TextPosition::new(pos.line + 1, 0))
        } else {
            ctx.document.len_chars()
        };
        let text = ctx.document.slice(TextRange::new(line_start, line_end));
        if let Some(ref mut cb) = self.clipboard {
            let _ = cb.set_text(text);
        }
        self.clipboard_is_line_copy = true;

fn handle_cut(&mut self, ctx: &mut CommandContext) {
    if let Some((start, end)) = ctx.view.selection_range() {
        // Selection active: cut selected text (existing behavior)
        let deleted_len = end - start;
        let text = ctx.document.slice(TextRange::new(start, end));
        if let Some(ref mut cb) = self.clipboard {
            let _ = cb.set_text(text.clone());
        }
        self.clipboard_is_line_copy = false;
        let op = EditOperation::delete(start, text);
        ctx.history.execute(op, ctx.document.buffer_mut());
        ctx.view.move_caret_to(start, false);
        ctx.view.clear_selection();
        ctx.document.update_syntax_incremental(start, deleted_len, 0);
        let affected = TextRange::new(start, end);
        self.emit_document_changed(ctx, Some(affected));
        self.emit_selection_changed(ctx);
    } else {
        // No selection: cut entire current line
        let pos = self.offset_to_position(ctx, ctx.view.caret_offset());
        let line_start = self.position_to_offset(ctx, TextPosition::new(pos.line, 0));
        let line_end = if pos.line + 1 < ctx.document.len_lines() {
            self.position_to_offset(ctx, TextPosition::new(pos.line + 1, 0))
        } else {
            ctx.document.len_chars()
        };
        let deleted_len = line_end - line_start;
        let text = ctx.document.slice(TextRange::new(line_start, line_end));
        if let Some(ref mut cb) = self.clipboard {
            let _ = cb.set_text(text.clone());
        }
        self.clipboard_is_line_copy = true;
        let op = EditOperation::delete(line_start, text);
        ctx.history.execute(op, ctx.document.buffer_mut());
        // Move caret to start of the (now current) line
        let new_offset = line_start.min(ctx.document.len_chars());
        ctx.view.move_caret_to(new_offset, false);
        ctx.view.clear_selection();
        ctx.document.update_syntax_incremental(line_start, deleted_len, 0);
        let affected = TextRange::new(line_start, line_end);
        self.emit_document_changed(ctx, Some(affected));
        self.emit_selection_changed(ctx);
    }
}

fn handle_paste(&mut self, ctx: &mut CommandContext) {
    let text = self.clipboard.as_mut().and_then(|cb| cb.get_text().ok());
    let is_line_copy = self.clipboard_is_line_copy;
    if let Some(text) = text {
        if is_line_copy && !ctx.view.has_selection() {
            // Line paste: insert above current line
            let pos = self.offset_to_position(ctx, ctx.view.caret_offset());
            let line_start = self.position_to_offset(ctx, TextPosition::new(pos.line, 0));
            // Ensure text ends with newline
            let paste_text = if text.ends_with('\n') {
                text
            } else {
                format!("{}\n", text)
            };
            let paste_len = paste_text.chars().count();
            let op = EditOperation::insert_string(line_start, paste_text);
            ctx.history.execute(op, ctx.document.buffer_mut());
            // Move caret to same column on the new (pushed-down) line
            let new_offset = line_start + paste_len + pos.column;
            let new_offset = new_offset.min(ctx.document.len_chars());
            ctx.view.move_caret_to(new_offset, false);
            ctx.view.clear_selection();
            ctx.document.update_syntax_incremental(line_start, 0, paste_len);
            let affected = TextRange::new(line_start, line_start + paste_len);
            self.emit_document_changed(ctx, Some(affected));
            self.emit_selection_changed(ctx);
        } else {
            // Normal paste (inline at cursor, replacing selection if any)
            if ctx.view.has_selection() {
                self.handle_delete_selection(ctx);
            }

}

5. Implement `auto_indent_paste`:

```rust
/// Adjusts the indentation of pasted text to match the current line's indentation.
///
/// Algorithm:
/// 1. Find the indentation of the line at the cursor (reference indent)
/// 2. Find the indentation of the first line of the clipboard text (source indent)
/// 3. For each line in the clipboard text, replace source indent with reference indent
///
/// Only adjusts if the paste text contains multiple lines. Single-line pastes
/// are inserted as-is.
fn auto_indent_paste(&self, ctx: &CommandContext, text: &str) -> String {
    // Single-line paste: no adjustment needed
    if !text.contains('\n') {
        return text.to_string();
    }

    // Get reference indentation from current line
    let pos = self.offset_to_position(ctx, ctx.view.caret_offset());
    let current_line = ctx.document.line(pos.line);
    let ref_indent: String = current_line
        .chars()
        .take_while(|c| *c == ' ' || *c == '\t')
        .collect();

    // Get source indentation from first non-empty line of paste text
    let first_content_line = text.lines().find(|l| !l.trim().is_empty());
    let src_indent: String = first_content_line
        .map(|l| l.chars().take_while(|c| *c == ' ' || *c == '\t').collect())
        .unwrap_or_default();

    // If source and reference are the same, no adjustment needed
    if ref_indent == src_indent {
        return text.to_string();
    }

    // Adjust each line
    let lines: Vec<&str> = text.split('\n').collect();
    let adjusted: Vec<String> = lines.iter().enumerate().map(|(i, line)| {
        if i == 0 {
            // First line inherits cursor position, don't adjust
            line.to_string()
        } else if line.trim().is_empty() {
            // Empty lines: keep as-is
            line.to_string()
        } else if line.starts_with(&src_indent) {
            // Replace source indent with reference indent
            format!("{}{}", ref_indent, &line[src_indent.len()..])
        } else {
            line.to_string()
        }
    }).collect();

    adjusted.join("\n")
}
```
