```rust
/// Character class for VS Code-style word boundaries.
/// Three classes: word (alphanumeric + underscore), punctuation, whitespace.
fn char_class(ch: char) -> u8 {
    if ch.is_alphanumeric() || ch == '_' { 0 }      // word
    else if ch.is_whitespace() || ch == '\n' { 2 }   // whitespace
    else { 1 }                                         // punctuation
}

/// Returns the (start, end) offsets of the word at the given offset.
/// Uses VS Code word boundary rules: alphanumeric+underscore runs are words,
/// punctuation runs are separate, whitespace runs are separate.
pub fn word_boundary_at(&self, offset: usize) -> (usize, usize) {
    let len = self.len_chars();
    if len == 0 { return (0, 0); }
    let offset = offset.min(len.saturating_sub(1));

    let ch = self.char_at(offset).unwrap_or(' ');
    let class = char_class(ch);

    // Scan backward to find word start
    let mut start = offset;
    while start > 0 {
        if let Some(prev_ch) = self.char_at(start - 1) {
            if char_class(prev_ch) != class { break; }
        }
        start -= 1;
    }

    // Scan forward to find word end
    let mut end = offset + 1;
    while end < len {
        if let Some(next_ch) = self.char_at(end) {
            if char_class(next_ch) != class { break; }
        }
        end += 1;
    }

    (start, end)
}
```

```rust
fn handle_drag_to(&self, ctx: &mut CommandContext, line: usize, col: usize, snap_mode: u32) {
    let clamped_line = line.min(ctx.document.len_lines().saturating_sub(1));
    let clamped_col = col.min(self.line_len(ctx, clamped_line));
    let offset = self.position_to_offset(ctx, TextPosition::new(clamped_line, clamped_col));

    match snap_mode {
        0 => {
            // Character-level drag: extend selection to exact offset
            ctx.view.move_caret_to(offset, true);
        }
        1 => {
            // Word-snap drag: extend selection to word boundary
            let (word_start, word_end) = ctx.document.buffer().word_boundary_at(offset);
            let anchor = ctx.view.selections.primary().anchor;
            // If dragging forward from anchor, extend to word_end
            // If dragging backward, extend to word_start
            if offset >= anchor {
                ctx.view.move_caret_to(word_end, true);
            } else {
                ctx.view.move_caret_to(word_start, true);
            }
        }
        2 => {
            // Line-snap drag: extend selection to full line boundaries
            let line_start = self.position_to_offset(ctx, TextPosition::new(clamped_line, 0));
            let anchor_line = self.offset_to_position(ctx, ctx.view.selections.primary().anchor).line;
            if clamped_line >= anchor_line {
                // Dragging down: extend to end of dragged line (include newline)
                let next_line_start = if clamped_line + 1 < ctx.document.len_lines() {
                    self.position_to_offset(ctx, TextPosition::new(clamped_line + 1, 0))
                } else {
                    ctx.document.len_chars()
                };
                ctx.view.move_caret_to(next_line_start, true);
            } else {
                // Dragging up: extend to start of dragged line
                ctx.view.move_caret_to(line_start, true);
            }
        }
        _ => {
            ctx.view.move_caret_to(offset, true);
        }
    }

    self.emit_selection_changed(ctx);
}
```
