  <name>Task 1: Add transition access to PaintContext</name>
  <files>ora/src/element/trait_def.rs</files>
  <action>
  Add methods to PaintContext for accessing the TransitionRegistry:

  ```rust
  use crate::animation::transition::{TransitionId, TransitionState, TransitionConfig};

  impl PaintContext {
      /// Get or create a TransitionState for the given element.
      /// Uses RefCell on AppContext.transition_registry for interior mutability.
      pub fn advance_transition_bg(
          &self,
          id: TransitionId,
          target: Color,
          config: &TransitionConfig,
      ) -> Color {
          let app_cx = unsafe { &*self.app_context };
          let mut registry = app_cx.transition_registry.borrow_mut();
          let state = registry.get_or_create(id);
          state.advance_bg(target, config)
      }

      pub fn advance_transition_opacity(
          &self,
          id: TransitionId,
          target: f32,
          config: &TransitionConfig,
      ) -> f32 {
          let app_cx = unsafe { &*self.app_context };
          let mut registry = app_cx.transition_registry.borrow_mut();
          let state = registry.get_or_create(id);
          state.advance_opacity(target, config)
      }

      pub fn advance_transition_color(
          &self,
          id: TransitionId,
          target: Color,
          config: &TransitionConfig,
      ) -> Color {
          let app_cx = unsafe { &*self.app_context };
          let mut registry = app_cx.transition_registry.borrow_mut();
          let state = registry.get_or_create(id);
          state.advance_color(target, config)
      }
  }
  ```

  The `self.app_context` is a `*const AppContext`. Because `transition_registry` is a `RefCell`, we can borrow_mut() through the const reference — RefCell provides interior mutability. This is safe because:
  1. The PaintContext is only used during paint phase (single-threaded)
  2. No other code borrows the registry during paint
  3. The RefCell runtime check catches any violation

  Do NOT change `*const AppContext` to `*mut AppContext` — keep the existing pattern.

     ```rust
     /// Assign a stable transition identity. Required for transitions to work.
     /// Views should assign unique IDs per element site (e.g., const per Div in render()).
     pub fn transition_id(mut self, id: TransitionId) -> Self {
         self.transition_id = Some(id);
         self
     }

     /// Animate background color changes with the given duration in ms.
     pub fn transition_bg(mut self, duration_ms: u32) -> Self {
         self.transition_spec
             .get_or_insert_with(TransitionSpec::default)
             .bg = Some(TransitionConfig {
                 duration_ms,
                 easing: Easing::EaseOut,
             });
         self
     }

     /// Animate opacity changes.
     pub fn transition_opacity(mut self, duration_ms: u32) -> Self {
         self.transition_spec
             .get_or_insert_with(TransitionSpec::default)
             .opacity = Some(TransitionConfig {
                 duration_ms,
                 easing: Easing::EaseOut,
             });
         self
     }

     /// Animate text/border color changes.
     pub fn transition_color(mut self, duration_ms: u32) -> Self {
         self.transition_spec
             .get_or_insert_with(TransitionSpec::default)
             .color = Some(TransitionConfig {
                 duration_ms,
                 easing: Easing::EaseOut,
             });
         self
     }

     /// Override easing for the most recently configured transition property.
     pub fn easing(mut self, easing: Easing) -> Self {
         if let Some(spec) = &mut self.transition_spec {
             // Apply to the last-configured property
             if let Some(c) = &mut spec.color { c.easing = easing; }
             else if let Some(o) = &mut spec.opacity { o.easing = easing; }
             else if let Some(b) = &mut spec.bg { b.easing = easing; }
         }
         self
     }
     ```

  3. Modify `Div::paint()` to use transitions:

     Find the section where hover_bg/active_bg are applied (the instant state switch). Replace with:

     ```rust
     // Determine target background based on interaction state
     let base_bg = match &self.style.background {
         Background::Solid(c) => *c,
         _ => Color::TRANSPARENT,
     };

     let target_bg = if cx.is_active(hitbox_id) {
         self.active_bg.unwrap_or(base_bg)
     } else if cx.is_hovered(hitbox_id) {
         self.hover_bg.unwrap_or(base_bg)
     } else {
         base_bg
     };

     // Apply transition if configured, otherwise instant
     let final_bg = if let (Some(tid), Some(spec)) = (self.transition_id, &self.transition_spec) {
         if let Some(bg_config) = &spec.bg {
             cx.advance_transition_bg(tid, target_bg, bg_config)
         } else {
             target_bg
         }
     } else {
         target_bg
     };

     style.background = Background::Solid(final_bg);
     ```

     Keep the existing non-transition path working — if no transition_id or transition_spec is set, behavior is exactly as before (instant switch).
