**Element implementation:**
```rust
pub struct DivState {
    layout_id: LayoutId,
}

impl Element for Div {
    type RequestLayoutState = DivState;

    fn request_layout(&mut self, cx: &mut LayoutContext) -> (LayoutId, DivState) {
        let id = cx.request_layout(&self.style);
        // Register children
        for child in &mut self.children {
            let child_id = child.request_layout(cx);
            cx.add_child(id, child_id);
        }
        (id, DivState { layout_id: id })
    }

    fn prepaint(&mut self, state: &mut DivState, cx: &mut PrepaintContext) {
        for child in &mut self.children {
            child.prepaint(cx);
        }
    }

    fn paint(&mut self, state: &mut DivState, cx: &mut PaintContext) {
        let bounds = cx.bounds(state.layout_id);
        // Emit PaintCommand::StyledRect with style properties and computed bounds
        cx.paint_styled_rect(&self.style, &bounds);
        // Paint children
        for child in &mut self.children {
            child.paint(cx);
        }
    }
}
```

**Element implementation:**
```rust
pub struct TextState {
    layout_id: LayoutId,
    // Buffer is stored for reuse between measure and paint (RESEARCH.md Pitfall 4)
    // We store the buffer index rather than the buffer itself to avoid ownership issues
    measured_size: Size<f32>,
}

impl Element for TextElement {
    type RequestLayoutState = TextState;

    fn request_layout(&mut self, cx: &mut LayoutContext) -> (LayoutId, TextState) {
        let id = cx.request_layout(&self.style);

        // Measure text to determine intrinsic size
        // The TextSystem will be accessed through the context
        // For now, set intrinsic size based on text measurement
        let max_width = match self.style.width {
            Length::Px(w) => Some(w),
            _ => if self.wrap { None /* will be set after layout */ } else { None },
        };

        // Text measurement happens via LayoutContext which has access to TextSystem
        let measured = cx.measure_text(&self.content, self.font_size, self.line_height, max_width);
        cx.set_intrinsic_size(id, measured);

        (id, TextState {
            layout_id: id,
            measured_size: measured,
        })
    }

    fn prepaint(&mut self, _state: &mut TextState, _cx: &mut PrepaintContext) {
        // No hitbox registration for text in Phase 2
    }

    fn paint(&mut self, state: &mut TextState, cx: &mut PaintContext) {
        let bounds = cx.bounds(state.layout_id);
        // Emit text paint command
        cx.paint_text(&self.content, self.font_size, self.line_height, &self.color, &bounds, self.wrap);
    }
}
```

  pub fn measure_text(&mut self, content: &str, font_size: f32, line_height: f32, max_width: Option<f32>) -> Size<f32> {
      // If TextSystem is available, use it for accurate measurement
      // Fallback: rough estimate based on font_size * char count
      if let Some(text_system) = &mut self.text_system {

          Size::new(width, line_height)
