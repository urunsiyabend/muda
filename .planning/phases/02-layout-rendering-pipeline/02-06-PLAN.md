   ```rust
   pub fn render_frame(
       &mut self,
       commands: &[PaintCommand],
   ) -> Result<(), wgpu::SurfaceError> {
       let output = self.surface.get_current_texture()?;
       let view = output.texture.create_view(&Default::default());
       let mut encoder = self.device.create_command_encoder(&Default::default());

       // Phase 1: Convert PaintCommands to GPU data
       let mut rect_instances: Vec<RectInstance> = Vec::new();
       for cmd in commands {
           match cmd {
               PaintCommand::StyledRect { bounds, style } => {
                   rect_instances.push(RectInstance::from_style(style, bounds, self.size));
               }
               PaintCommand::Text { content, font_size, line_height, color, bounds, wrap } => {
                   let max_width = if *wrap { Some(bounds.size.width) } else { None };
                   let (buffer, _) = self.text_system.measure_text(content, *font_size, *line_height, max_width);
                   self.text_system.add_text_area(
                       buffer,
                       bounds.origin.x,
                       bounds.origin.y,
                       *bounds,
                       Color::from_array(*color),
                   );
               }
               PaintCommand::Rect { x, y, width, height, color } => {
                   // Legacy Phase 1 rect support
                   rect_instances.push(RectInstance::from_legacy(*x, *y, *width, *height, *color, self.size));
               }
           }
       }

       // Phase 2: Prepare GPU data
       self.rect_renderer.prepare(&self.device, &self.queue, &rect_instances);
       self.text_system.prepare(&self.device, &self.queue, self.size.0, self.size.1)
           .expect("Text prepare failed");

       // Phase 3: Render pass
       {
           let mut render_pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
               label: Some("ora-render-pass"),
               color_attachments: &[Some(wgpu::RenderPassColorAttachment {
                   view: &view,
                   resolve_target: None,
                   ops: wgpu::Operations {
                       load: wgpu::LoadOp::Clear(wgpu::Color { r: 0.1, g: 0.1, b: 0.12, a: 1.0 }),
                       store: wgpu::StoreOp::Store,
                   },
               })],
               depth_stencil_attachment: None,
               timestamp_writes: None,
               occlusion_query_set: None,
           });

           // Draw rectangles (backgrounds, borders, shadows)
           self.rect_renderer.render(&mut render_pass);

           // Draw text on top
           self.text_system.render(&mut render_pass)
               .expect("Text render failed");
       }

       self.queue.submit(std::iter::once(encoder.finish()));
       output.present();

       // Clear text system for next frame
       self.text_system.clear();

       Ok(())
   }
   ```
