      if span.style == TextStyle::Selection {

  2. **Rewrite render_selection_rects()** to use `selection_ranges` instead of scanning spans:
     ```rust
     fn render_selection_bg_layer(&self, cx: &mut ViewContext) -> AnyElement {
         let theme = cx.theme();
         let selection_color = theme.color(ColorToken::Selection);
         let mut line_elements: Vec<AnyElement> = Vec::new();

         for (line_index, line) in self.visible_lines.iter().enumerate() {
             if line.selection_ranges.is_empty() {
                 // Empty placeholder to maintain vertical alignment
                 line_elements.push(
                     Div::new().w(pct(100.0)).h(px(LINE_HEIGHT)).shrink(0.0).into()
                 );
                 continue;
             }
             // For each selection range on this line, create a background rect.
             // For simplicity, use a single rect covering the full selected region.
             // If multiple ranges exist, merge or handle separately.
             let (start_col, end_col) = line.selection_ranges[0];

             // For multi-line selection: if end_col == usize::MAX or similar sentinel,
             // extend to full line width. Check the convention from core_editor.
             // Simple approach: calculate pixel positions.
             let total_line_chars: usize = line.spans.iter()
                 .map(|s| s.text.chars().count())
                 .sum();

             // If selection extends to end of line in a multi-line selection,
             // extend to full width for edge-to-edge rendering
             let effective_end = if end_col >= total_line_chars {
                 // Edge-to-edge: use a large width to fill the line
                 // The overflow_hidden on the parent will clip it
                 (total_line_chars + 20).max(end_col) // extend past text
             } else {
                 end_col
             };

             let start_px = start_col as f32 * self.char_width;
             let width_px = (effective_end - start_col) as f32 * self.char_width;

             // Build a line-height div with the selection rect inside
             let selection_rect = Div::new()
                 .w(px(width_px))
                 .h(px(LINE_HEIGHT))
                 .bg(selection_color);

             // Use a row with left padding to position the selection rect
             let line_div = Div::new()
                 .flex_row()
                 .w(pct(100.0))
                 .h(px(LINE_HEIGHT))
                 .shrink(0.0)
                 .pl(start_px)
                 .child(selection_rect);

             line_elements.push(line_div.into());
         }

         Div::new()
             .flex_col()
             .w(pct(100.0))
             .children(line_elements)
             .into()
     }
     ```

             .child(inner_text)
             .into();
