use crate::events::types::Modifiers;

/// Platform-agnostic key identifier.
/// Wraps winit's Key type for ora's abstraction layer.
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub enum Key {
    /// Named keys (Enter, Escape, Tab, etc.)
    Named(NamedKey),
    /// Character keys (a-z, 0-9, symbols)
    Character(String),
}

/// Named keys that don't produce characters.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum NamedKey {
    Enter,
    Escape,
    Tab,
    Backspace,
    Delete,
    Insert,
    Home,
    End,
    PageUp,
    PageDown,
    ArrowUp,
    ArrowDown,
    ArrowLeft,
    ArrowRight,
    F1, F2, F3, F4, F5, F6, F7, F8, F9, F10, F11, F12,
    Space,
}

/// A keystroke combines a key with modifier state.
/// This is the unit of keybinding matching.
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Keystroke {
    pub key: Key,
    pub modifiers: Modifiers,
}

impl Keystroke {
    pub fn new(key: Key, modifiers: Modifiers) -> Self {
        Self { key, modifiers }
    }

    /// Create keystroke from character with modifiers.
    pub fn char(c: &str, ctrl: bool, alt: bool, shift: bool) -> Self {
        Self {
            key: Key::Character(c.to_lowercase()),
            modifiers: Modifiers { ctrl, alt, shift, meta: false },
        }
    }

    /// Create keystroke from named key with modifiers.
    pub fn named(key: NamedKey, ctrl: bool, alt: bool, shift: bool) -> Self {
        Self {
            key: Key::Named(key),
            modifiers: Modifiers { ctrl, alt, shift, meta: false },
        }
    }
}

/// Keyboard event from winit, translated to ora types.
#[derive(Debug, Clone)]
pub struct KeyboardEvent {
    pub keystroke: Keystroke,
    /// Whether this is a key repeat event
    pub repeat: bool,
    /// Text produced by this key (if any)
    pub text: Option<String>,
}

/// Convert winit KeyEvent to ora KeyboardEvent.
pub fn translate_key_event(
    event: &winit::event::KeyEvent,
    modifiers: &Modifiers,
) -> Option<KeyboardEvent> {
    let key = match &event.logical_key {
        winit::keyboard::Key::Named(named) => {
            let named_key = match named {
                winit::keyboard::NamedKey::Enter => NamedKey::Enter,
                winit::keyboard::NamedKey::Escape => NamedKey::Escape,
                winit::keyboard::NamedKey::Tab => NamedKey::Tab,
                winit::keyboard::NamedKey::Backspace => NamedKey::Backspace,
                winit::keyboard::NamedKey::Delete => NamedKey::Delete,
                winit::keyboard::NamedKey::Insert => NamedKey::Insert,
                winit::keyboard::NamedKey::Home => NamedKey::Home,
                winit::keyboard::NamedKey::End => NamedKey::End,
                winit::keyboard::NamedKey::PageUp => NamedKey::PageUp,
                winit::keyboard::NamedKey::PageDown => NamedKey::PageDown,
                winit::keyboard::NamedKey::ArrowUp => NamedKey::ArrowUp,
                winit::keyboard::NamedKey::ArrowDown => NamedKey::ArrowDown,
                winit::keyboard::NamedKey::ArrowLeft => NamedKey::ArrowLeft,
                winit::keyboard::NamedKey::ArrowRight => NamedKey::ArrowRight,
                winit::keyboard::NamedKey::F1 => NamedKey::F1,
                winit::keyboard::NamedKey::F2 => NamedKey::F2,
                winit::keyboard::NamedKey::F3 => NamedKey::F3,
                winit::keyboard::NamedKey::F4 => NamedKey::F4,
                winit::keyboard::NamedKey::F5 => NamedKey::F5,
                winit::keyboard::NamedKey::F6 => NamedKey::F6,
                winit::keyboard::NamedKey::F7 => NamedKey::F7,
                winit::keyboard::NamedKey::F8 => NamedKey::F8,
                winit::keyboard::NamedKey::F9 => NamedKey::F9,
                winit::keyboard::NamedKey::F10 => NamedKey::F10,
                winit::keyboard::NamedKey::F11 => NamedKey::F11,
                winit::keyboard::NamedKey::F12 => NamedKey::F12,
                winit::keyboard::NamedKey::Space => NamedKey::Space,
                _ => return None, // Unsupported named key
            };
            Key::Named(named_key)
        }
        winit::keyboard::Key::Character(c) => {
            Key::Character(c.to_lowercase())
        }
        _ => return None,
    };

    let keystroke = Keystroke {
        key,
        modifiers: *modifiers,
    };

    let text = event.text.as_ref().map(|s| s.to_string());

    Some(KeyboardEvent {
        keystroke,
        repeat: event.repeat,
        text,
    })
}

use crate::events::keyboard::Keystroke;
use std::any::{Any, TypeId};
use std::collections::HashMap;

/// Trait for typed actions that can be triggered by keybindings.
/// Actions are unit structs or structs with data that describe what to do.
pub trait Action: 'static + Clone + Send + Sync {
    /// Human-readable name for debugging/display.
    fn name(&self) -> &'static str;

    /// Unique type identifier for this action type.
    fn type_id(&self) -> TypeId {
        TypeId::of::<Self>()
    }

    /// Box clone for type erasure.
    fn boxed_clone(&self) -> Box<dyn Action>;

    /// Downcast to concrete type.
    fn as_any(&self) -> &dyn Any;
}

/// Helper macro to implement Action trait for simple action structs.
#[macro_export]
macro_rules! define_action {
    ($name:ident) => {
        #[derive(Debug, Clone)]
        pub struct $name;

        impl $crate::events::actions::Action for $name {
            fn name(&self) -> &'static str {
                stringify!($name)
            }

            fn boxed_clone(&self) -> Box<dyn $crate::events::actions::Action> {
                Box::new(self.clone())
            }

            fn as_any(&self) -> &dyn std::any::Any {
                self
            }
        }
    };
}

/// Context for keybinding matching.
/// Allows bindings to be scoped to specific element types or states.
#[derive(Debug, Clone, Default)]
pub struct KeyContext {
    /// Set of active context identifiers (e.g., "TextInput", "Editor", "Dialog")
    pub identifiers: Vec<String>,
}

impl KeyContext {
    pub fn new() -> Self {
        Self { identifiers: Vec::new() }
    }

    pub fn add(&mut self, identifier: impl Into<String>) {
        self.identifiers.push(identifier.into());
    }

    pub fn has(&self, identifier: &str) -> bool {
        self.identifiers.iter().any(|id| id == identifier)
    }
}

/// A single keybinding: keystroke -> action, optionally scoped to context.
pub struct KeyBinding {
    pub keystroke: Keystroke,
    pub action: Box<dyn Action>,
    /// Optional context predicate - binding only matches if predicate returns true.
    pub context_predicate: Option<Box<dyn Fn(&KeyContext) -> bool + Send + Sync>>,
}

impl KeyBinding {
    pub fn new(keystroke: Keystroke, action: impl Action) -> Self {
        Self {
            keystroke,
            action: Box::new(action),
            context_predicate: None,
        }
    }

    pub fn when(mut self, predicate: impl Fn(&KeyContext) -> bool + Send + Sync + 'static) -> Self {
        self.context_predicate = Some(Box::new(predicate));
        self
    }

    /// Check if this binding matches the given keystroke and context.
    pub fn matches(&self, keystroke: &Keystroke, context: &KeyContext) -> bool {
        if self.keystroke != *keystroke {
            return false;
        }

        match &self.context_predicate {
            Some(pred) => pred(context),
            None => true,
        }
    }
}

/// Registry of keybindings.
/// Matches keystrokes to actions with context awareness.
pub struct Keymap {
    bindings: Vec<KeyBinding>,
}

impl Keymap {
    pub fn new() -> Self {
        Self { bindings: Vec::new() }
    }

    /// Add a keybinding.
    pub fn bind(&mut self, binding: KeyBinding) {
        self.bindings.push(binding);
    }

    /// Add a simple keybinding without context.
    pub fn bind_key(&mut self, keystroke: Keystroke, action: impl Action) {
        self.bind(KeyBinding::new(keystroke, action));
    }

    /// Find the matching action for a keystroke in a given context.
    /// Later bindings override earlier ones (last match wins).
    pub fn match_action(&self, keystroke: &Keystroke, context: &KeyContext) -> Option<Box<dyn Action>> {
        // Iterate in reverse - last binding wins
        self.bindings
            .iter()
            .rev()
            .find(|b| b.matches(keystroke, context))
            .map(|b| b.action.boxed_clone())
    }
}

impl Default for Keymap {
    fn default() -> Self {
        Self::new()
    }
}

/// Type alias for action handlers.
/// Handler receives the action and returns whether it was handled.
pub type ActionHandler = Box<dyn FnMut(&dyn Action) -> bool>;

/// Registry of action handlers keyed by TypeId.
pub struct ActionRegistry {
    handlers: HashMap<TypeId, Vec<ActionHandler>>,
}

impl ActionRegistry {
    pub fn new() -> Self {
        Self { handlers: HashMap::new() }
    }

    /// Register a handler for a specific action type.
    pub fn on_action<A: Action>(&mut self, mut handler: impl FnMut(&A) -> bool + 'static) {
        let type_id = TypeId::of::<A>();
        let wrapped: ActionHandler = Box::new(move |action: &dyn Action| {
            if let Some(typed) = action.as_any().downcast_ref::<A>() {
                handler(typed)
            } else {
                false
            }
        });

        self.handlers
            .entry(type_id)
            .or_insert_with(Vec::new)
            .push(wrapped);
    }

    /// Dispatch an action to registered handlers.
    /// Returns true if any handler processed the action.
    pub fn dispatch(&mut self, action: &dyn Action) -> bool {
        let type_id = action.type_id();
        if let Some(handlers) = self.handlers.get_mut(&type_id) {
            for handler in handlers.iter_mut() {
                if handler(action) {
                    return true;
                }
            }
        }
        false
    }

    pub fn clear(&mut self) {
        self.handlers.clear();
    }
}

impl Default for ActionRegistry {
    fn default() -> Self {
        Self::new()
    }
}

       // Only process key press (not release)
       if event.state != winit::event::ElementState::Pressed {
           return;
       }

       // Handle Tab for focus navigation (before action matching)
       if let winit::keyboard::Key::Named(winit::keyboard::NamedKey::Tab) = event.logical_key {

               self.app_context.focus_prev();

               self.app_context.focus_next();
