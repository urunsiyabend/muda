use crate::events::mouse::{HitboxId, MouseButton};
use crate::events::dispatch::EventHandlers;
use std::collections::HashSet;

/// Mouse capture state for drag operations.
#[derive(Debug, Clone)]
pub struct MouseCapture {
    /// The element that captured the mouse.
    pub hitbox_id: HitboxId,
    /// The button that was pressed to capture.
    pub button: MouseButton,
}

/// Framework-managed interaction state.
/// Tracks hover, active (pressed), and mouse capture.
pub struct InteractionState {
    /// Currently hovered hitbox (topmost under cursor).
    hovered: Option<HitboxId>,
    /// Set of all hitboxes in the hover path (target + ancestors).
    hover_path: HashSet<HitboxId>,
    /// Currently active (pressed) hitbox.
    active: Option<HitboxId>,
    /// Set of all hitboxes in the active path (target + ancestors).
    active_path: HashSet<HitboxId>,
    /// Mouse capture for drag operations.
    mouse_capture: Option<MouseCapture>,
}

impl InteractionState {
    pub fn new() -> Self {
        Self {
            hovered: None,
            hover_path: HashSet::new(),
            active: None,
            active_path: HashSet::new(),
            mouse_capture: None,
        }
    }

    /// Update hover state based on hit test result.
    /// Builds hover path including all ancestors.
    pub fn update_hover(
        &mut self,
        hitbox_id: Option<HitboxId>,
        handlers: &EventHandlers,
    ) {
        self.hovered = hitbox_id;
        self.hover_path.clear();

        if let Some(id) = hitbox_id {
            // Add target and all ancestors to hover path
            self.hover_path.insert(id);
            let mut current = id;
            while let Some(&parent) = handlers.parent_map.get(&current) {
                self.hover_path.insert(parent);
                current = parent;
            }
        }
    }

    /// Set active state on mouse down.
    /// Builds active path including all ancestors.
    pub fn set_active(
        &mut self,
        hitbox_id: HitboxId,
        handlers: &EventHandlers,
    ) {
        self.active = Some(hitbox_id);
        self.active_path.clear();

        // Add target and all ancestors to active path
        self.active_path.insert(hitbox_id);
        let mut current = hitbox_id;
        while let Some(&parent) = handlers.parent_map.get(&current) {
            self.active_path.insert(parent);
            current = parent;
        }
    }

    /// Clear active state on mouse up.
    pub fn clear_active(&mut self) {
        self.active = None;
        self.active_path.clear();
    }

    /// Check if a hitbox is in the hover path (hovered or ancestor of hovered).
    pub fn is_hovered(&self, id: HitboxId) -> bool {
        self.hover_path.contains(&id)
    }

    /// Check if a hitbox is in the active path (active or ancestor of active).
    pub fn is_active(&self, id: HitboxId) -> bool {
        self.active_path.contains(&id)
    }

    /// Get the topmost hovered hitbox.
    pub fn hovered_hitbox(&self) -> Option<HitboxId> {
        self.hovered
    }

    /// Get the active hitbox.
    pub fn active_hitbox(&self) -> Option<HitboxId> {
        self.active
    }

    /// Capture mouse for drag operations.
    /// While captured, all mouse events route to the capturing element.
    pub fn capture_mouse(&mut self, hitbox_id: HitboxId, button: MouseButton) {
        self.mouse_capture = Some(MouseCapture { hitbox_id, button });
    }

    /// Release mouse capture.
    pub fn release_mouse_capture(&mut self) {
        self.mouse_capture = None;
    }

    /// Get current mouse capture.
    pub fn mouse_capture(&self) -> Option<&MouseCapture> {
        self.mouse_capture.as_ref()
    }

    /// Check if mouse is captured.
    pub fn is_mouse_captured(&self) -> bool {
        self.mouse_capture.is_some()
    }

    /// Clear all hover/active state (called on window deactivation).
    pub fn clear_all(&mut self) {
        self.hovered = None;
        self.hover_path.clear();
        self.active = None;
        self.active_path.clear();
        self.mouse_capture = None;
    }
}

impl Default for InteractionState {
    fn default() -> Self {
        Self::new()
    }
}

       let point = Point { x: position.x as f32, y: position.y as f32 };

       // Determine hit target (respect mouse capture)
       let hit_target = if let Some(capture) = self.app_context.interaction_state.mouse_capture() {

           hit_test(&self.hitboxes, point)

       // Update hover state
       self.app_context.interaction_state.update_hover(hit_target, &self.event_handlers);

       // Dispatch mouse move event
       if let Some(hit_id) = hit_target {
           let event = MouseMoveEvent {

               modifiers: self.modifiers,

           dispatch_mouse_move(&event, hit_id, &mut self.event_handlers);

       // Request redraw for hover effects
       if let Some(gpu_state) = &self.gpu_state {

       }
   }

   WindowEvent::MouseInput { state, button, .. } => {
       let mouse_button = /* existing conversion */;

       // Determine target (respect mouse capture)
       let hit_target = if let Some(capture) = self.app_context.interaction_state.mouse_capture() {

                   self.app_context.interaction_state.set_active(hit_id, &self.event_handlers);

                   let event = MouseDownEvent {
                       position: self.cursor_position,
                       button: mouse_button,
                       modifiers: self.modifiers,

                   dispatch_mouse_down(&event, hit_id, &mut self.event_handlers);
               }
               winit::event::ElementState::Released => {
                   // Clear active state
                   self.app_context.interaction_state.clear_active();

                   // Release capture if button matches

                       if capture.button == mouse_button {
                           self.app_context.interaction_state.release_mouse_capture();

                       position: self.cursor_position,
                       button: mouse_button,
                       modifiers: self.modifiers,
                   };
                   dispatch_mouse_up(&event, hit_id, &mut self.event_handlers);
               }
           }
       }

       // Request redraw for active state changes
       if let Some(gpu_state) = &self.gpu_state {
           gpu_state.window.request_redraw();
       }
   }

//! Interactive demo showcasing ora's event system.
//!
//! Features demonstrated:
//! - Hover effects (background color change)
//! - Active/pressed state (darker on click)
//! - Focus management with keyboard navigation (Tab/Shift+Tab)
//! - Focus ring on keyboard-focused elements
//! - Typed actions with keybindings
//!
//! Run with: cargo run --example interactive_demo

use ora::{
    App, AppContext, View, ViewContext, WindowContext,
    Div, TextElement, Model, Subscription,
    Style, Color, FocusHandle, FocusId,
    define_action, Keystroke, NamedKey, Key,
};
use ora::events::mouse::HitboxId;

// Define typed actions
define_action!(IncrementAction);
define_action!(DecrementAction);
define_action!(ResetAction);

/// Demo application state

    counter: i32,
}

/// Interactive button component

    label: String,
    focus_handle: FocusHandle,
    hitbox_id: Option<HitboxId>,
    on_click: Box<dyn Fn(&mut AppContext)>,
}

/// Main demo view
struct DemoView {
    state: Model<DemoState>,
    buttons: Vec<InteractiveButton>,
    _subscription: Subscription,
}

impl DemoView {
    fn new(cx: &mut ViewContext) -> Self {
        let state = cx.new_model(DemoState { counter: 0 });

        // Observe state changes to trigger re-render
        let state_clone = state.clone();
        let subscription = cx.observe(&state, move |cx| {
            log::debug!("State changed, counter = {}", cx.read_model(&state_clone).counter);
        });
        subscription.detach();

        // Register keybindings
        cx.bind_key(
            Keystroke::named(NamedKey::ArrowUp, false, false, false),
            IncrementAction,
        );
        cx.bind_key(
            Keystroke::named(NamedKey::ArrowDown, false, false, false),
            DecrementAction,
        );
        cx.bind_key(
            Keystroke::char("r", true, false, false), // Ctrl+R
            ResetAction,
        );

        // Register action handlers
        let state_for_inc = state.clone();
        cx.on_action(move |_action: &IncrementAction| {
            // Can't access cx here, need different pattern
            log::info!("Increment action triggered");
            true
        });

        Self {
            state,
            buttons: Vec::new(), // Buttons created during render
            _subscription: Subscription::new(0, ora::subscription::CleanupAction::RemoveObserver {
                observed: ora::effect::EntityId::from_raw_parts(0, 0),
