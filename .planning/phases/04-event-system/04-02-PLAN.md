   ```rust

       Keyboard,  // Tab, arrow keys
       Mouse,     // Click
       Programmatic, // Code-driven focus
   }
   ```

3. **FocusHandle** - reference-counted handle that survives re-renders:
   ```rust
   use std::sync::Arc;

   #[derive(Clone)]
   pub struct FocusHandle {
       pub(crate) id: FocusId,
       /// Reference counting - focus is "alive" while any handle exists
       _ref: Arc<()>,
   }

   impl FocusHandle {
       pub fn id(&self) -> FocusId {
           self.id
       }
   }
   ```

4. **FocusState** - centralized focus tracking:
   ```rust
   pub struct FocusState {
       /// Currently focused element
       focused_id: Option<FocusId>,
       /// How the current focus was gained
       focus_source: FocusSource,
       /// Counter for generating unique FocusIds
       next_focus_id: u64,
       /// Ordered list of focusable elements (built during prepaint)
       focus_order: Vec<FocusId>,
   }

   impl FocusState {
       pub fn new() -> Self {
           Self {
               focused_id: None,
               focus_source: FocusSource::Programmatic,
               next_focus_id: 0,
               focus_order: Vec::new(),
           }
       }

       /// Create a new FocusHandle with unique ID
       pub fn create_focus_handle(&mut self) -> FocusHandle {
           let id = FocusId(self.next_focus_id);
           self.next_focus_id += 1;
           FocusHandle {
               id,
               _ref: Arc::new(()),
           }
       }

       /// Set focus to a specific element
       pub fn set_focused(&mut self, id: FocusId, source: FocusSource) {
           self.focused_id = Some(id);
           self.focus_source = source;
       }

       /// Clear focus
       pub fn clear_focus(&mut self) {
           self.focused_id = None;
       }

       /// Get currently focused element
       pub fn focused_id(&self) -> Option<FocusId> {
           self.focused_id
       }

       /// Check if focus came from keyboard
       pub fn is_keyboard_focused(&self) -> bool {
           self.focused_id.is_some() && self.focus_source == FocusSource::Keyboard
       }

       /// Clear focus order (called at start of prepaint)
       pub fn clear_focus_order(&mut self) {
           self.focus_order.clear();
       }

       /// Register a focusable element in tab order
       pub fn register_focusable(&mut self, id: FocusId) {
           self.focus_order.push(id);
       }

       /// Move focus to next element (Tab)
       pub fn focus_next(&mut self) {
           if self.focus_order.is_empty() {
               return;
           }

           let current_idx = self.focused_id
               .and_then(|id| self.focus_order.iter().position(|&fid| fid == id));

           let next_idx = match current_idx {
               Some(idx) => (idx + 1) % self.focus_order.len(),
               None => 0,
           };

           if let Some(&next_id) = self.focus_order.get(next_idx) {
               self.set_focused(next_id, FocusSource::Keyboard);
           }
       }

       /// Move focus to previous element (Shift+Tab)
       pub fn focus_prev(&mut self) {
           if self.focus_order.is_empty() {
               return;
           }

           let current_idx = self.focused_id
               .and_then(|id| self.focus_order.iter().position(|&fid| fid == id));

           let prev_idx = match current_idx {
               Some(0) => self.focus_order.len() - 1,
               Some(idx) => idx - 1,
               None => self.focus_order.len() - 1,
           };

           if let Some(&prev_id) = self.focus_order.get(prev_idx) {
               self.set_focused(prev_id, FocusSource::Keyboard);
           }
       }
   }
   ```

   /// Call this in view constructor, store the handle.
   pub fn focus_handle(&mut self) -> FocusHandle {
       self.focus_state.create_focus_handle()
   }

   /// Check if a specific element is focused.
   pub fn is_focused(&self, id: FocusId) -> bool {
       self.focus_state.focused_id() == Some(id)
   }

   /// Check if focus was gained via keyboard (for focus ring styling).
   pub fn focus_visible(&self, id: FocusId) -> bool {
       self.is_focused(id) && self.focus_state.is_keyboard_focused()

   pub fn focus(&mut self, handle: &FocusHandle) {
       self.focus_state.set_focused(handle.id, FocusSource::Programmatic);
   }

       self.focus_state.set_focused(handle.id, FocusSource::Mouse);
