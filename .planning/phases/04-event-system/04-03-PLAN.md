use crate::events::mouse::{HitboxId, MouseDownEvent, MouseUpEvent, MouseMoveEvent};
use std::cell::Cell;

/// Phase of event dispatch (DOM-like two-phase model).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DispatchPhase {
    /// Capture phase: event flows from root to target.
    Capture,
    /// Bubble phase: event flows from target to root.
    Bubble,
}

/// Context provided to event handlers during dispatch.
/// Allows handlers to stop propagation or access event metadata.
pub struct EventContext {
    /// Current dispatch phase.
    phase: DispatchPhase,
    /// Whether propagation has been stopped.
    propagation_stopped: Cell<bool>,
    /// The target hitbox that was hit.
    target: HitboxId,
}

impl EventContext {
    pub fn new(target: HitboxId, phase: DispatchPhase) -> Self {
        Self {
            phase,
            propagation_stopped: Cell::new(false),
            target,
        }
    }

    /// Get the current dispatch phase.
    pub fn phase(&self) -> DispatchPhase {
        self.phase
    }

    /// Stop event propagation. No further handlers will be called.
    pub fn stop_propagation(&self) {
        self.propagation_stopped.set(true);
    }

    /// Check if propagation has been stopped.
    pub fn is_propagation_stopped(&self) -> bool {
        self.propagation_stopped.get()
    }

    /// Get the target hitbox ID.
    pub fn target(&self) -> HitboxId {
        self.target
    }
}

/// Type alias for mouse event handlers.
pub type MouseDownHandler = Box<dyn FnMut(&MouseDownEvent, &EventContext)>;
pub type MouseUpHandler = Box<dyn FnMut(&MouseUpEvent, &EventContext)>;
pub type MouseMoveHandler = Box<dyn FnMut(&MouseMoveEvent, &EventContext)>;

/// Storage for registered event handlers keyed by HitboxId.
pub struct EventHandlers {
    pub mouse_down: Vec<(HitboxId, MouseDownHandler)>,
    pub mouse_up: Vec<(HitboxId, MouseUpHandler)>,
    pub mouse_move: Vec<(HitboxId, MouseMoveHandler)>,
    /// Parent chain: maps child HitboxId to parent HitboxId for bubbling
    pub parent_map: std::collections::HashMap<HitboxId, HitboxId>,
}

impl EventHandlers {
    pub fn new() -> Self {
        Self {
            mouse_down: Vec::new(),
            mouse_up: Vec::new(),
            mouse_move: Vec::new(),
            parent_map: std::collections::HashMap::new(),
        }
    }

    pub fn clear(&mut self) {
        self.mouse_down.clear();
        self.mouse_up.clear();
        self.mouse_move.clear();
        self.parent_map.clear();
    }
}

impl Default for EventHandlers {
    fn default() -> Self {
        Self::new()
    }
}

/// Build the path from root to target for two-phase dispatch.
/// Returns vec in order [root, ..., parent, target].
pub fn build_dispatch_path(
    target: HitboxId,
    parent_map: &std::collections::HashMap<HitboxId, HitboxId>,
) -> Vec<HitboxId> {
    let mut path = vec![target];
    let mut current = target;

    while let Some(&parent) = parent_map.get(&current) {
        path.push(parent);
        current = parent;
    }

    path.reverse(); // Now [root, ..., target]
    path
}

/// Dispatch a mouse down event through the element tree.
pub fn dispatch_mouse_down(
    event: &MouseDownEvent,
    target: HitboxId,
    handlers: &mut EventHandlers,
) {
    let path = build_dispatch_path(target, &handlers.parent_map);

    // Capture phase: root to target
    for &hitbox_id in &path {
        let ctx = EventContext::new(target, DispatchPhase::Capture);
        for (id, handler) in &mut handlers.mouse_down {
            if *id == hitbox_id {
                handler(event, &ctx);
                if ctx.is_propagation_stopped() {
                    return;
                }
            }
        }
    }

    // Bubble phase: target to root
    for &hitbox_id in path.iter().rev() {
        let ctx = EventContext::new(target, DispatchPhase::Bubble);
        for (id, handler) in &mut handlers.mouse_down {
            if *id == hitbox_id {
                handler(event, &ctx);
                if ctx.is_propagation_stopped() {
                    return;
                }
            }
        }
    }
}

   impl PrepaintContext {
       /// Register a mouse down handler for a hitbox.
       pub fn on_mouse_down(
           &mut self,
           hitbox_id: HitboxId,
           handler: impl FnMut(&crate::events::mouse::MouseDownEvent, &crate::events::dispatch::EventContext) + 'static,
       ) {
           self.event_handlers.mouse_down.push((hitbox_id, Box::new(handler)));
       }

       /// Register a mouse up handler for a hitbox.
       pub fn on_mouse_up(
           &mut self,
           hitbox_id: HitboxId,
           handler: impl FnMut(&crate::events::mouse::MouseUpEvent, &crate::events::dispatch::EventContext) + 'static,
       ) {
           self.event_handlers.mouse_up.push((hitbox_id, Box::new(handler)));
       }

       /// Register a mouse move handler for a hitbox.
       pub fn on_mouse_move(
           &mut self,
           hitbox_id: HitboxId,
           handler: impl FnMut(&crate::events::mouse::MouseMoveEvent, &crate::events::dispatch::EventContext) + 'static,
       ) {
           self.event_handlers.mouse_move.push((hitbox_id, Box::new(handler)));
       }

       /// Push a hitbox onto the parent stack (call before processing children).
       pub fn push_hitbox_parent(&mut self, id: HitboxId) {
           self.hitbox_stack.push(id);
       }

       /// Pop a hitbox from the parent stack (call after processing children).
       pub fn pop_hitbox_parent(&mut self) {
           self.hitbox_stack.pop();
       }

       /// Register parent-child relationship for current hitbox.
       /// Call this when registering a hitbox to establish the parent chain.
       pub fn register_hitbox_with_parent(&mut self, bounds: crate::style::units::Rect, opaque: bool) -> HitboxId {
           let id = self.register_hitbox(bounds, opaque);

           // Register parent relationship if we have a parent
           if let Some(&parent) = self.hitbox_stack.last() {
               self.event_handlers.parent_map.insert(id, parent);
           }

           id
       }

       /// Extract event handlers for dispatch.
       pub(crate) fn take_event_handlers(&mut self) -> EventHandlers {
           std::mem::take(&mut self.event_handlers)
       }
   }
