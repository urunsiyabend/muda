```rust
/// Result of a fuzzy match attempt.
#[derive(Clone, Debug)]
pub struct FuzzyMatch {
    /// Match quality score (higher = better match).
    pub score: i32,
    /// Character indices in the target string that matched query characters.
    pub match_positions: Vec<usize>,
}

/// Fuzzy match a query against a target string.
///
/// Returns None if not all query characters found in sequence.
/// Returns Some(FuzzyMatch) with score and match positions on success.
/// Scoring: +10 for consecutive matches, +5 for word boundary matches, +1 otherwise.
pub fn fuzzy_match(query: &str, target: &str) -> Option<FuzzyMatch> {
    if query.is_empty() {
        return Some(FuzzyMatch { score: 0, match_positions: vec![] });
    }

    let query_chars: Vec<char> = query.to_lowercase().chars().collect();
    let target_chars: Vec<char> = target.to_lowercase().chars().collect();

    let mut match_positions = Vec::new();
    let mut qi = 0;
    let mut score = 0i32;

    for (ti, &tc) in target_chars.iter().enumerate() {
        if qi < query_chars.len() && tc == query_chars[qi] {
            // Bonus for consecutive matches
            if let Some(&last) = match_positions.last() {
                if ti == last + 1 {
                    score += 10;
                } else {
                    score += 1;
                }
            } else {
                score += 1;
            }
            // Bonus for word boundary (start, after space/underscore)
            if ti == 0 || matches!(target_chars.get(ti.wrapping_sub(1)), Some(' ' | '_' | '-')) {
                score += 5;
            }
            match_positions.push(ti);
            qi += 1;
        }
    }

    if qi == query_chars.len() {
        Some(FuzzyMatch { score, match_positions })
    } else {
        None
    }
}
```

```rust
pub struct CommandItem {
    /// Unique command identifier.
    pub id: String,
    /// Display label (e.g., "Open File").
    pub label: String,
    /// Optional keyboard shortcut display (e.g., "Ctrl+O").
    pub shortcut: Option<String>,
}

impl CommandItem {
    pub fn new(id: impl Into<String>, label: impl Into<String>) -> Self {
        Self { id: id.into(), label: label.into(), shortcut: None }
    }

    pub fn with_shortcut(mut self, shortcut: impl Into<String>) -> Self {
        self.shortcut = Some(shortcut.into());
        self
    }
}
```

```rust
impl View for CommandPaletteView {
    fn render(&self, cx: &mut ViewContext) -> AnyElement {
        if !self.visible {
            return Div::new().w(px(0.0)).h(px(0.0)).into();
        }

        let theme = cx.theme();

        // Backdrop (semi-transparent, optional — or just the palette floating)
        // For CommandPalette, no backdrop needed — it's an overlay, not a modal

        // Positioning container: centers palette horizontally, offsets from top
        // Use a full-screen flex_col + align_center + pt for top offset
        let positioning = Div::new()
            .flex_col()
            .w(pct(100.0))
            .h(pct(100.0))
            .align_center()
            .pt(PALETTE_TOP_OFFSET);

        // Palette container
        let mut palette = Div::new()
            .flex_col()
            .w(px(PALETTE_WIDTH))
            .bg(theme.color(ColorToken::BgElevated))
            .border(1.0, theme.color(ColorToken::Border))
            .border_radius(PALETTE_BORDER_RADIUS)
            .shadow(0.0, 8.0, 32.0, 0.0, Color::rgba(0.0, 0.0, 0.0, 0.5));

        // Input row showing query text
        palette = palette.child(self.render_input(cx));

        // Separator line
        palette = palette.child(
            Div::new()
                .h(px(1.0))
                .bg(theme.color(ColorToken::Border))
        );

        // Result list (fixed height, shows up to VISIBLE_ROWS)
        let list_height = VISIBLE_ROWS as f32 * ROW_HEIGHT;
        let mut result_list = Div::new()
            .flex_col()
            .h(px(list_height))
            .overflow_hidden();

        for (i, (cmd_idx, fuzzy)) in self.filtered.iter().take(VISIBLE_ROWS).enumerate() {
            let cmd = &self.commands[*cmd_idx];
            let is_selected = i == self.selected_index;
            result_list = result_list.child(self.render_result(cmd, fuzzy, is_selected, cx));
        }

        palette = palette.child(result_list);
        positioning.child(palette).into()
    }
}
```

```rust
fn render_highlighted_label(&self, label: &str, positions: &[usize], cx: &mut ViewContext) -> Div {
    let theme = cx.theme();
    let mut row = Div::new().flex_row().align_center();
    let chars: Vec<char> = label.chars().collect();
    let mut pos_set: std::collections::HashSet<usize> = positions.iter().copied().collect();

    // Build segments: consecutive non-matched chars as one TextElement,
    // consecutive matched chars as another TextElement with Accent color
    let mut i = 0;
    while i < chars.len() {
        let is_match = pos_set.contains(&i);
        let start = i;
        while i < chars.len() && pos_set.contains(&i) == is_match {
            i += 1;
        }
        let segment: String = chars[start..i].iter().collect();
        let color = if is_match {
            theme.color(ColorToken::Accent)
        } else {
            theme.color(ColorToken::FgPrimary)
        };
        row = row.child(TextElement::new(&segment).size(13.0).color(color));
    }
    row
}
```
