```rust
/// Panel types available in the IDE.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum PanelKind {
    Output,
    Problems,
    Terminal,
    Debug,
}

impl PanelKind {
    pub fn label(&self) -> &'static str {
        match self {
            Self::Output => "Output",
            Self::Problems => "Problems",
            Self::Terminal => "Terminal",
            Self::Debug => "Debug",
        }
    }
}

/// Preset height positions for double-click cycling.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum PanelPreset {
    Collapsed,  // height = 0 (hidden)
    Half,       // height = 200px
    Full,       // height = 400px
}

/// Panel layout state.
pub struct PanelState {
    /// Current panel height in pixels.
    pub height: f32,
    /// Whether panel is collapsed (height = 0).
    pub is_collapsed: bool,
    /// Preferred height when expanding from collapsed.
    pub preferred_height: f32,
    /// Currently active panel tab.
    pub active_tab: PanelKind,
    /// Tab order (reorderable by drag).
    pub tab_order: Vec<PanelKind>,
}

const DEFAULT_PANEL_HEIGHT: f32 = 200.0;
const MIN_PANEL_HEIGHT: f32 = 100.0;
const MAX_PANEL_HEIGHT: f32 = 500.0;
const RESIZE_HANDLE_HEIGHT: f32 = 4.0;  // Visual: 1px, Hitbox: 4px (RESEARCH Pitfall 3)

const PANEL_TAB_FONT_SIZE: f32 = 12.0;
```

View rendering:
```rust
impl View for PanelManagerView {
    fn render(&self, cx: &mut ViewContext) -> AnyElement {
        if self.state.is_collapsed {
            return Div::new().w(pct(100.0)).h(px(0.0)).into();
        }

        let theme = cx.theme();

        let mut container = Div::new()
            .flex_col()
            .w(pct(100.0))
            .h(px(self.state.height))
            .bg(theme.color(ColorToken::BgSecondary));

        // Resize handle at top (thin visible line, wider hitbox area)
        container = container.child(self.render_resize_handle(cx));

        // Tab bar row
        container = container.child(self.render_tab_bar(cx));

        // Content area (grows to fill remaining space)
        container = container.child(self.render_content(cx));

        container.into()
    }
}
```

Helper methods:

**render_resize_handle**: A thin horizontal Div that serves as the drag handle.
```rust
fn render_resize_handle(&self, cx: &mut ViewContext) -> Div {
    let theme = cx.theme();
    // Visual: 1px border line. Hitbox will be wider (handled by prepaint in future).
    // For now, render a 4px tall Div with centered 1px line for the handle.
    Div::new()
        .w(pct(100.0))
        .h(px(RESIZE_HANDLE_HEIGHT))
        .flex_col()
        .justify_center()
        .hover_bg(theme.color(ColorToken::Accent))  // Blue highlight on hover
        .child(
            Div::new()
                .w(pct(100.0))
                .h(px(1.0))
                .bg(theme.color(ColorToken::Border))
        )
}
```

Note on drag: Full drag-to-resize requires mouse capture integration which works at the event loop level. For now, the resize handle renders visually correct and has hover feedback. Actual drag tracking will be wired when PanelManager is integrated into AppLayout (Plan 08) or can be done via the app's event handling. **render_tab_bar**: Row of tab buttons for each panel kind.
```rust
fn render_tab_bar(&self, cx: &mut ViewContext) -> Div {
    let theme = cx.theme();
    let mut row = Div::new()
        .flex_row()
        .h(px(PANEL_TAB_HEIGHT))
        .w(pct(100.0))
        .bg(theme.color(ColorToken::BgSecondary))
        .border(1.0, theme.color(ColorToken::Border));  // bottom border only ideally

    for kind in &self.state.tab_order {
        let is_active = *kind == self.state.active_tab;
        let tab_bg = if is_active {
            theme.color(ColorToken::BgPrimary)
        } else {
            Color::transparent()
        };
        let tab_text_color = if is_active {
            theme.color(ColorToken::FgPrimary)
        } else {
            theme.color(ColorToken::FgSecondary)
        };
        let bottom_border_color = if is_active {
            theme.color(ColorToken::Accent)
        } else {
            Color::transparent()
        };

        let tab = Div::new()
            .flex_row()
            .align_center()
            .h(px(PANEL_TAB_HEIGHT))
            .px(PANEL_TAB_PADDING_H)
            .bg(tab_bg)
            .hover_bg(theme.color(ColorToken::BgElevated))
            .child(
                TextElement::new(kind.label())
                    .size(PANEL_TAB_FONT_SIZE)
                    .color(tab_text_color)
            );

        row = row.child(tab);
    }

    row
}
```

```rust
fn render_content(&self, cx: &mut ViewContext) -> Div {
    let theme = cx.theme();

        PanelKind::Output => "Output panel content",
        PanelKind::Problems => "No problems detected",
        PanelKind::Terminal => "Terminal ready",
        PanelKind::Debug => "Debug console",
    };

    Div::new()
        .flex_col()
        .grow(1.0)
        .p(PANEL_CONTENT_PADDING)
        .bg(theme.color(ColorToken::BgPrimary))
        .child(
            TextElement::new(content_text)
                .size(13.0)
                .color(theme.color(ColorToken::FgMuted))
        )
}
```
