AppLayout struct:
```rust
/// Top-level application layout orchestrator.
///
/// Manages the arrangement of all UI regions:
/// - Sidebar (left, collapsible)
/// - Tab bar (top of editor area)
/// - Editor area (gutter + text area)
/// - Bottom panel (collapsible, resizable)
/// - Status bar (bottom)
///
/// Overlays (rendered via Stack z-layering):
/// - CommandPalette
/// - ContextMenu
/// - Toast notifications
/// - Dialog (modal)
pub struct AppLayout {
    // Child views (owned by AppLayout)
    pub sidebar: SidebarView,
    pub tab_bar: TabBarView,
    pub gutter: GutterView,
    pub text_area: TextAreaView,
    pub status_bar: StatusBarView,
    pub panel_manager: PanelManagerView,
    pub command_palette: CommandPaletteView,
    pub dialog: DialogView,
    pub toast: Toast,
    pub context_menu: ContextMenu,

    // Layout state
    pub sidebar_visible: bool,
    pub panel_visible: bool,
}
```

View implementation:
```rust
impl View for AppLayout {
    fn render(&self, cx: &mut ViewContext) -> AnyElement {
        let theme = cx.theme();

        // === Build main layout (no overlays) ===

        // Editor area: gutter + text area in a row
        let editor_area = Div::new()
            .flex_row()
            .grow(1.0)
            .overflow_hidden()
            .child(self.gutter.render(cx))
            .child(
                Div::new()
                    .flex_col()
                    .grow(1.0)
                    .child(self.text_area.render(cx))
            );

        // Main column: tab bar, editor area, optional panel, status bar
        let mut main_col = Div::new()
            .flex_col()
            .grow(1.0)
            .bg(theme.color(ColorToken::BgPrimary));

        main_col = main_col.child(self.tab_bar.render(cx));
        main_col = main_col.child(editor_area);

        if self.panel_visible {
            main_col = main_col.child(self.panel_manager.render(cx));
        }

        main_col = main_col.child(self.status_bar.render(cx));

        // Root row: optional sidebar + main column
        let mut root_row = Div::new()
            .flex_row()
            .w(pct(100.0))
            .h(pct(100.0))
            .bg(theme.color(ColorToken::BgPrimary));

        if self.sidebar_visible {
            root_row = root_row.child(self.sidebar.render(cx));
        }

        root_row = root_row.child(main_col);

        // === Layer overlays using Stack ===

        let has_overlays = self.command_palette.is_visible()
            || self.context_menu.is_visible()
            || self.toast.has_notifications()
            || !matches!(&self.dialog_is_none(), true);  // dialog check

        // If there are overlays, wrap in Stack for z-layering
        // Layer order (bottom to top): main layout, context menu, command palette, dialog, toast
        let mut root_stack = stack()
            .w(pct(100.0))
            .h(pct(100.0))
            .child(root_row);

        // Context menu overlay (positioned at cursor)
        if self.context_menu.is_visible() {
            root_stack = root_stack.child(
                self.context_menu.render(cx)
            );
        }

        // Command palette overlay (centered near top)
        if self.command_palette.is_visible() {
            root_stack = root_stack.child(
                self.command_palette.render(cx)
            );
        }

        // Dialog overlay (modal with backdrop)
        // Dialog handles its own backdrop and centering
        root_stack = root_stack.child(self.dialog.render(cx));

        // Toast overlay (bottom-right, topmost layer)
        if self.toast.has_notifications() {
            // Position toast at bottom-right using a full-screen flex container
            let toast_container = Div::new()
                .flex_col()
                .w(pct(100.0))
                .h(pct(100.0))
                .justify_end()   // push to bottom
                .align_end()     // push to right
                .child(self.toast.render(cx));

            root_stack = root_stack.child(toast_container);
        }

        root_stack.into()
    }
}
```
