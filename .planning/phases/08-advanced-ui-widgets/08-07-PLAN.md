Create `ora/src/elements/context_menu.rs` implementing the ContextMenu widget (WIDGET-07).

ContextMenu is a popup menu that appears at a given position. It is NOT an Element in the three-phase lifecycle sense (since it needs absolute positioning which is handled by Stack). Instead, implement it as a View-like struct with a render method that returns a Div tree, similar to how DialogView works.

```rust
use crate::context::ViewContext;
use crate::elements::{Div, TextElement};
use crate::events::FocusHandle;
use crate::style::{px, Color};
use crate::theme::ColorToken;

const MENU_WIDTH: f32 = 200.0;
const MENU_ITEM_HEIGHT: f32 = 28.0;
const MENU_PADDING_V: f32 = 4.0;
const MENU_FONT_SIZE: f32 = 13.0;
const MENU_ITEM_PADDING_H: f32 = 12.0;
const MENU_BORDER_RADIUS: f32 = 4.0;
const SEPARATOR_HEIGHT: f32 = 1.0;
const SEPARATOR_MARGIN_V: f32 = 4.0;

/// A context menu item.
#[derive(Clone, Debug)]
pub enum ContextMenuItem {
    /// A clickable menu item with label and optional shortcut.
    Action {
        label: String,
        shortcut: Option<String>,
        disabled: bool,
        on_click: Option<Box<dyn Fn() + 'static>>,
    },
    /// A visual separator line.
    Separator,
}

impl ContextMenuItem {
    pub fn action(label: impl Into<String>) -> Self {
        Self::Action {
            label: label.into(),
            shortcut: None,
            disabled: false,
            on_click: None,
        }
    }

    pub fn with_shortcut(mut self, shortcut: impl Into<String>) -> Self {
        if let Self::Action { shortcut: ref mut s, .. } = self {
            *s = Some(shortcut.into());
        }
        self
    }

    pub fn disabled(mut self) -> Self {
        if let Self::Action { disabled: ref mut d, .. } = self {
            *d = true;
        }
        self
    }

    pub fn on_click(mut self, handler: impl Fn() + 'static) -> Self {
        if let Self::Action { on_click: ref mut h, .. } = self {
            *h = Some(Box::new(handler));
        }
        self
    }

    pub fn separator() -> Self {
        Self::Separator
    }
}
```

NOTE: on_click uses Box<dyn Fn()> which is not Clone. ContextMenuItem with on_click cannot derive Clone. Adjust: remove Clone from ContextMenuItem, or store handlers separately. Simplest: don't derive Clone, items are created fresh each render.

```rust
impl ContextMenu {
    pub fn render(&self, cx: &mut ViewContext) -> Div {
        if !self.visible {
            return Div::new().w(px(0.0)).h(px(0.0));
        }

        let theme = cx.theme();

        // Menu container
        // Position is handled by the parent (AppLayout wraps in Stack with absolute positioning)
        // This just renders the menu content
        let mut menu = Div::new()
            .flex_col()
            .w(px(MENU_WIDTH))
            .py(MENU_PADDING_V)
            .bg(theme.color(ColorToken::BgElevated))
            .border(1.0, theme.color(ColorToken::Border))
            .border_radius(MENU_BORDER_RADIUS)
            .shadow(0.0, 4.0, 16.0, 0.0, Color::rgba(0.0, 0.0, 0.0, 0.4));

        let mut item_index = 0;
        for item in &self.items {
            match item {
                ContextMenuItem::Action { label, shortcut, disabled, .. } => {
                    let is_selected = item_index == self.selected_index;
                    let bg = if is_selected && !*disabled {
                        theme.color(ColorToken::BgSecondary)
                    } else {
                        Color::transparent()
                    };
                    let text_color = if *disabled {
                        let mut c = theme.color(ColorToken::FgMuted);
                        c.a = 0.5;
                        c
                    } else {
                        theme.color(ColorToken::FgPrimary)
                    };

                    let mut row = Div::new()
                        .flex_row()
                        .justify_between()
                        .align_center()
                        .h(px(MENU_ITEM_HEIGHT))
                        .px(MENU_ITEM_PADDING_H)
                        .bg(bg);

                    if !*disabled {
                        row = row.hover_bg(theme.color(ColorToken::BgSecondary));
                    }

                    row = row.child(
                        TextElement::new(label)
                            .size(MENU_FONT_SIZE)
                            .color(text_color)
                    );

                    if let Some(sc) = shortcut {
                        row = row.child(
                            TextElement::new(sc)
                                .size(MENU_FONT_SIZE - 1.0)
                                .color(theme.color(ColorToken::FgMuted))
                        );
                    }

                    menu = menu.child(row);
                    item_index += 1;
                }
                ContextMenuItem::Separator => {
                    menu = menu.child(
                        Div::new()
                            .w(px(MENU_WIDTH - 8.0))  // inset
                            .h(px(SEPARATOR_HEIGHT))
                            .my(SEPARATOR_MARGIN_V)
                            .mx(4.0)
                            .bg(theme.color(ColorToken::Border))
                    );
                }
            }
        }

        menu
    }
}
```

```rust
pub fn clamp_to_viewport(&mut self, window_width: f32, window_height: f32) {
    let menu_height = self.estimated_height();
    if self.position_x + MENU_WIDTH > window_width {
        self.position_x = (self.position_x - MENU_WIDTH).max(0.0);
    }
    if self.position_y + menu_height > window_height {
        self.position_y = (self.position_y - menu_height).max(0.0);
    }
}

fn estimated_height(&self) -> f32 {
    let mut h = MENU_PADDING_V * 2.0;
    for item in &self.items {
        h += match item {
            ContextMenuItem::Action { .. } => MENU_ITEM_HEIGHT,
            ContextMenuItem::Separator => SEPARATOR_HEIGHT + SEPARATOR_MARGIN_V * 2.0,
        };
    }
    h
}
```

Create `ora/src/elements/toast.rs` implementing the Toast widget (WIDGET-08).

CONTEXT decisions:
- Info/success auto-dismiss after 3-5 seconds
- Warning/error persist until user dismisses
- Position: bottom-right (RESEARCH Pitfall 6: avoids conflicts with CommandPalette at top)

RESEARCH recommendation: Use `dismiss_at: Option<Instant>` instead of cx.spawn() for timer management. The event loop's about_to_wait hook checks if it's time to dismiss.

```rust
use crate::context::ViewContext;
use crate::elements::{Div, TextElement};
use crate::style::{px, Color};
use crate::theme::ColorToken;
use std::time::{Duration, Instant};

const TOAST_WIDTH: f32 = 320.0;
const TOAST_HEIGHT: f32 = 48.0;
const TOAST_PADDING_H: f32 = 16.0;
const TOAST_PADDING_V: f32 = 12.0;
const TOAST_BORDER_RADIUS: f32 = 6.0;
const TOAST_FONT_SIZE: f32 = 13.0;
const TOAST_MARGIN: f32 = 16.0;  // margin from viewport edge
const AUTO_DISMISS_DURATION: Duration = Duration::from_secs(4);  // 4 seconds for info/success

/// Toast notification severity level.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ToastSeverity {
    Info,
    Success,
    Warning,
    Error,
}

impl ToastSeverity {
    /// Whether this severity auto-dismisses.
    pub fn auto_dismiss(&self) -> bool {
        matches!(self, Self::Info | Self::Success)
    }

    /// Color token for the severity accent.
    pub fn color_token(&self) -> ColorToken {
        match self {
            Self::Info => ColorToken::Accent,
            Self::Success => ColorToken::Success,
            Self::Warning => ColorToken::Warning,
            Self::Error => ColorToken::Error,
        }
    }
}

/// A single toast notification.
#[derive(Clone, Debug)]
pub struct ToastNotification {
    pub message: String,
    pub severity: ToastSeverity,
    pub created_at: Instant,
    pub dismiss_at: Option<Instant>,
    pub dismissed: bool,
}

impl ToastNotification {
    pub fn new(message: impl Into<String>, severity: ToastSeverity) -> Self {
        let dismiss_at = if severity.auto_dismiss() {
            Some(Instant::now() + AUTO_DISMISS_DURATION)
        } else {
            None
        };
        Self {
            message: message.into(),
            severity,
            created_at: Instant::now(),
            dismiss_at,
            dismissed: false,
        }
    }

    /// Check if this toast should be dismissed now.
    pub fn should_dismiss(&self) -> bool {
        if self.dismissed { return true; }
        if let Some(dismiss_at) = self.dismiss_at {
            Instant::now() >= dismiss_at
        } else {
            false
        }
    }

    pub fn dismiss(&mut self) {
        self.dismissed = true;
    }
}

/// Toast container that manages multiple toast notifications.
///
/// Renders in bottom-right of viewport. Multiple toasts stack vertically.
pub struct Toast {
    notifications: Vec<ToastNotification>,
}

impl Toast {
    pub fn new() -> Self {
        Self { notifications: Vec::new() }
    }

    /// Add a new toast notification.
    pub fn show(&mut self, message: impl Into<String>, severity: ToastSeverity) {
        self.notifications.push(ToastNotification::new(message, severity));
    }

    /// Remove expired and dismissed toasts. Call from event loop about_to_wait.
    pub fn tick(&mut self) {
        self.notifications.retain(|t| !t.should_dismiss());
    }

    /// Dismiss the most recent persistent toast (for user click).
    pub fn dismiss_latest(&mut self) {
        if let Some(toast) = self.notifications.iter_mut().rev()
            .find(|t| !t.dismissed && t.dismiss_at.is_none()) {
            toast.dismiss();
        }
    }

    /// Whether there are any active notifications.
    pub fn has_notifications(&self) -> bool {
        !self.notifications.is_empty()
    }

    /// Render all active toasts positioned in bottom-right.
    pub fn render(&self, cx: &mut ViewContext) -> Div {
        if self.notifications.is_empty() {
            return Div::new().w(px(0.0)).h(px(0.0));
        }

        let theme = cx.theme();

        // Container positioned at bottom-right
        // Parent (AppLayout) handles absolute positioning via Stack
        let mut container = Div::new()
            .flex_col()
            .align_end()  // right-align
            .p(TOAST_MARGIN);

        for notification in &self.notifications {
            if notification.dismissed { continue; }

            let severity_color = theme.color(notification.severity.color_token());

            let toast_div = Div::new()
                .flex_row()
                .align_center()
                .w(px(TOAST_WIDTH))
                .py(TOAST_PADDING_V)
                .px(TOAST_PADDING_H)
                .mb(8.0)  // gap between stacked toasts
                .bg(theme.color(ColorToken::BgElevated))
                .border(1.0, theme.color(ColorToken::Border))
                .border_radius(TOAST_BORDER_RADIUS)
                .shadow(0.0, 4.0, 16.0, 0.0, Color::rgba(0.0, 0.0, 0.0, 0.3))
                // Left accent strip (colored by severity)
                .child(
                    Div::new()
                        .w(px(3.0))
                        .h(px(TOAST_HEIGHT - TOAST_PADDING_V * 2.0))
                        .bg(severity_color)
                        .border_radius(2.0)
                        .mr(12.0)
                )
                // Message text
                .child(
                    TextElement::new(&notification.message)
                        .size(TOAST_FONT_SIZE)
                        .color(theme.color(ColorToken::FgPrimary))
                )
                // Dismiss button for persistent toasts
                .child(if notification.dismiss_at.is_none() {
                    Div::new()
                        .flex_row()
                        .align_center()
                        .justify_center()
                        .w(px(20.0))
                        .h(px(20.0))
                        .ml(8.0)
                        .hover_bg(theme.color(ColorToken::BgSecondary))
                        .border_radius(3.0)
                        .child(
                            TextElement::new("x")
                                .size(12.0)
                                .color(theme.color(ColorToken::FgMuted))
                        )
                } else {
                    Div::new().w(px(0.0))
                });

            container = container.child(toast_div);
        }

        container
    }
}
```
