```rust
/// A node in the file tree hierarchy.
#[derive(Clone, Debug)]
pub struct FileTreeNode {
    /// Display name (file or directory name).
    pub name: String,
    /// File extension (e.g., "rs", "js") for icon coloring. Empty for directories.
    pub extension: String,
    /// Whether this is a directory.
    pub is_dir: bool,
    /// Whether this directory is expanded (only meaningful for directories).
    pub is_expanded: bool,
    /// Nested children (only for directories).
    pub children: Vec<FileTreeNode>,
}

impl FileTreeNode {
    pub fn file(name: impl Into<String>, extension: impl Into<String>) -> Self {
        Self { name: name.into(), extension: extension.into(), is_dir: false, is_expanded: false, children: vec![] }
    }

    pub fn dir(name: impl Into<String>, expanded: bool, children: Vec<FileTreeNode>) -> Self {
        Self { name: name.into(), extension: String::new(), is_dir: true, is_expanded: expanded, children }
    }
}

/// Presentation data for the file tree.
#[derive(Clone, Debug, Default)]
pub struct FileTreePresentation {
    /// Root-level nodes.
    pub roots: Vec<FileTreeNode>,
    /// Index of the currently selected entry in the flattened list.
    pub selected_index: Option<usize>,
}
```

The key rendering approach: Flatten the tree into a list of (depth, node) pairs for rendering. Only include children of expanded directories.

```rust
impl FileTreeView {
    /// Flatten tree into renderable rows with depth information.
    fn flatten_nodes(nodes: &[FileTreeNode], depth: usize, out: &mut Vec<(usize, &FileTreeNode)>) {
        for node in nodes {
            out.push((depth, node));
            if node.is_dir && node.is_expanded {
                Self::flatten_nodes(&node.children, depth + 1, out);
            }
        }
    }

    fn render_row(&self, node: &FileTreeNode, depth: usize, flat_index: usize, cx: &mut ViewContext) -> Div {
        let theme = cx.theme();
        let is_selected = self.presentation.selected_index == Some(flat_index);
        let indent = depth as f32 * INDENT_WIDTH;

        let bg = if is_selected {
            theme.color(ColorToken::Accent)  // with low alpha for selection
            // Actually use a muted accent: theme.color(ColorToken::Selection)
        } else {
            Color::transparent()
        };

        let mut row = Div::new()
            .flex_row()
            .align_center()
            .h(px(TREE_ROW_HEIGHT))
            .px(ROW_PADDING_H)
            .bg(bg)
            .hover_bg(theme.color(ColorToken::BgElevated));

        // Indent spacer
        if indent > 0.0 {
            row = row.child(Div::new().w(px(indent)).shrink(0.0));
        }

        // Indent guides: thin vertical lines for each depth level
        // (Rendered as part of the indent spacer area)
        // For simplicity, render guides as part of the indent area above

        // Chevron (directories only)
        let chevron_text = if node.is_dir {
            if node.is_expanded { "v " } else { "> " }
        } else {
            "  "  // spacer for alignment
        };
        row = row.child(
            TextElement::new(chevron_text)
                .size(LABEL_FONT_SIZE)
                .color(theme.color(ColorToken::FgMuted))
        );

        // File icon (colored dot/square based on extension)
        let icon_char = if node.is_dir { "D " } else { "F " };  // simple markers
        let icon_color = if node.is_dir {
            theme.color(ColorToken::Warning)  // amber for folders
        } else {
            icon_color_for_extension(&node.extension)
        };
        row = row.child(
            TextElement::new(icon_char)
                .size(LABEL_FONT_SIZE)
                .color(icon_color)
        );

        // File/directory name
        let name_color = if is_selected {
            theme.color(ColorToken::FgPrimary)
        } else {
            theme.color(ColorToken::FgPrimary)
        };
        row = row.child(
            TextElement::new(&node.name)
                .size(LABEL_FONT_SIZE)
                .color(name_color)
        );

        row
    }
}
```

View implementation:
```rust
impl View for FileTreeView {
    fn render(&self, cx: &mut ViewContext) -> AnyElement {
        let mut rows = Vec::new();
        Self::flatten_nodes(&self.presentation.roots, 0, &mut rows);

        let theme = cx.theme();
        let mut container = Div::new()
            .flex_col()
            .grow(1.0)
            .bg(theme.color(ColorToken::BgSecondary));

        for (flat_index, (depth, node)) in rows.iter().enumerate() {
            container = container.child(self.render_row(node, *depth, flat_index, cx));
        }

        container.into()
    }
}
```
