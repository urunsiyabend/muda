//! Element Library Demo
//!
//! Demonstrates Phase 05 features:
//! - Builder API with px()/pct() unit functions
//! - Button variants (Primary, Secondary, Ghost, Destructive)
//! - Stack container for z-layering
//! - Image element with placeholder

use ora::{
    App, View, ViewContext, Element, AnyElement, Model,
    Div, TextElement, Button, button, ButtonVariant, Stack, stack, Image, img, ObjectFit,
    px, pct, Color, JustifyContent, AlignItems, FocusHandle,
};
use std::sync::Arc;

fn main() {
    env_logger::init();

    let app = App::new();
    let model = app.new_model(DemoState::default());
    let view = DemoView::new(model.clone());
    let window = app.open_window(view, "Element Library Demo", 900, 700);

    ora::run(app);
}

#[derive(Default)]
struct DemoState {
    selected_variant: usize,
}

struct DemoView {
    model: Model<DemoState>,
    focus_handles: [FocusHandle; 4], // One for each button variant
}

impl DemoView {
    fn new(model: Model<DemoState>) -> Self {
        Self {
            model,
            focus_handles: std::array::from_fn(|_| FocusHandle::new()),
        }
    }
}

impl View for DemoView {
    fn render(&mut self, cx: &mut ViewContext) -> AnyElement {
        // Main container
        Div::new()
            .w(pct(100.0))
            .h(pct(100.0))
            .bg(Color::rgb(0.12, 0.12, 0.14))
            .p(24.0)
            .flex_col()
            .gap(24.0)
            // Title
            .child(
                TextElement::new("Element Library Demo")
                    .size(24.0)
                    .color(Color::white())
            )
            // Section 1: Builder API Demo
            .child(self.render_builder_api_section())
            // Section 2: Button Variants Demo
            .child(self.render_button_variants_section())
            // Section 3: Stack Demo
            .child(self.render_stack_section())
            // Section 4: Image Demo
            .child(self.render_image_section())
            .into()
    }
}

impl DemoView {
    fn render_builder_api_section(&self) -> AnyElement {
        Div::new()
            .flex_col()
            .gap(12.0)
            .child(
                TextElement::new("Builder API with Units")
                    .size(18.0)
                    .color(Color::rgb(0.8, 0.8, 0.85))
            )
            .child(
                Div::new()
                    .row()
                    .gap(16.0)
                    // Fixed width box with px()
                    .child(
                        Div::new()
                            .w(px(120.0))
                            .h(px(80.0))
                            .bg(Color::rgb(0.2, 0.5, 0.8))
                            .border_radius(8.0)
                            .justify_center()
                            .align_center()
                            .child(
                                TextElement::new("px(120)")
                                    .size(12.0)
                                    .color(Color::white())
                            )
                    )
                    // Percentage width box
                    .child(
                        Div::new()
                            .w(pct(30.0))
                            .h(px(80.0))
                            .bg(Color::rgb(0.5, 0.2, 0.8))
                            .border_radius(8.0)
                            .justify_center()
                            .align_center()
                            .child(
                                TextElement::new("pct(30)")
                                    .size(12.0)
                                    .color(Color::white())
                            )
                    )
                    // CSS-style alignment demo
                    .child(
                        Div::new()
                            .w(px(200.0))
                            .h(px(80.0))
                            .bg(Color::rgb(0.8, 0.5, 0.2))
                            .border_radius(8.0)
                            .justify(JustifyContent::SpaceBetween)
                            .items(AlignItems::Center)
                            .px(12.0)
                            .child(
                                TextElement::new("Start")
                                    .size(12.0)
                                    .color(Color::white())
                            )
                            .child(
                                TextElement::new("End")
                                    .size(12.0)
                                    .color(Color::white())
                            )
                    )
            )
            .into()
    }

    fn render_button_variants_section(&self) -> AnyElement {
        Div::new()
            .flex_col()
            .gap(12.0)
            .child(
                TextElement::new("Button Variants")
                    .size(18.0)
                    .color(Color::rgb(0.8, 0.8, 0.85))
            )
            .child(
                Div::new()
                    .row()
                    .gap(16.0)
                    .child(
                        button("Primary")
                            .primary()
                            .focusable(self.focus_handles[0].clone())
                            .on_click(|| log::info!("Primary clicked"))
                    )
                    .child(
                        button("Secondary")
                            .secondary()
                            .focusable(self.focus_handles[1].clone())
                            .on_click(|| log::info!("Secondary clicked"))
                    )
                    .child(
                        button("Ghost")
                            .ghost()
                            .focusable(self.focus_handles[2].clone())
                            .on_click(|| log::info!("Ghost clicked"))
                    )
                    .child(
                        button("Destructive")
                            .destructive()
                            .focusable(self.focus_handles[3].clone())
                            .on_click(|| log::info!("Destructive clicked"))
                    )
            )
            .child(
                TextElement::new("Hover over buttons to see state changes")
                    .size(12.0)
                    .color(Color::rgb(0.5, 0.5, 0.55))
            )
            .into()
    }

    fn render_stack_section(&self) -> AnyElement {
        Div::new()
            .flex_col()
            .gap(12.0)
            .child(
                TextElement::new("Stack Container (Z-Layering)")
                    .size(18.0)
                    .color(Color::rgb(0.8, 0.8, 0.85))
            )
            .child(
                stack()
                    .w(300.0)
                    .h(150.0)
                    // Background layer
                    .child(
                        Div::new()
                            .w(px(300.0))
                            .h(px(150.0))
                            .bg(Color::rgb(0.3, 0.3, 0.35))
                            .border_radius(12.0)
                    )
                    // Middle layer - offset card
                    .child(
                        Div::new()
                            .w(px(200.0))
                            .h(px(100.0))
                            .bg(Color::rgb(0.4, 0.2, 0.6))
                            .border_radius(8.0)
                            .m(20.0)
                    )
                    // Top layer - small accent
                    .child(
                        Div::new()
                            .w(px(80.0))
                            .h(px(40.0))
                            .bg(Color::rgb(0.8, 0.6, 0.2))
                            .border_radius(4.0)
                            .m(40.0)
                    )
            )
            .child(
                TextElement::new("Children layer: background -> middle -> top")
                    .size(12.0)
                    .color(Color::rgb(0.5, 0.5, 0.55))
            )
            .into()
    }

    fn render_image_section(&self) -> AnyElement {
        Div::new()
            .flex_col()
            .gap(12.0)
            .child(
                TextElement::new("Image Element (Placeholder)")
                    .size(18.0)
                    .color(Color::rgb(0.8, 0.8, 0.85))
            )
            .child(
                Div::new()
                    .row()
                    .gap(16.0)
                    // Contain fit
                    .child(
                        Div::new()
                            .flex_col()
                            .gap(4.0)
                            .child(
                                img("nonexistent.png")
                                    .w(120.0)
                                    .h(80.0)
                                    .object_fit(ObjectFit::Contain)
                                    .placeholder_color(Color::rgb(0.2, 0.4, 0.3))
                            )
                            .child(
                                TextElement::new("Contain")
                                    .size(12.0)
                                    .color(Color::rgb(0.6, 0.6, 0.65))
                            )
                    )
                    // Cover fit
                    .child(
                        Div::new()
                            .flex_col()
                            .gap(4.0)
                            .child(
                                img("nonexistent.png")
                                    .w(120.0)
                                    .h(80.0)
                                    .object_fit(ObjectFit::Cover)
                                    .placeholder_color(Color::rgb(0.4, 0.2, 0.3))
                            )
                            .child(
                                TextElement::new("Cover")
                                    .size(12.0)
                                    .color(Color::rgb(0.6, 0.6, 0.65))
                            )
                    )
                    // Fill fit
                    .child(
                        Div::new()
                            .flex_col()
                            .gap(4.0)
                            .child(
                                img("nonexistent.png")
                                    .w(120.0)
                                    .h(80.0)
                                    .object_fit(ObjectFit::Fill)
                                    .placeholder_color(Color::rgb(0.3, 0.2, 0.4))
                            )
                            .child(
                                TextElement::new("Fill")
                                    .size(12.0)
                                    .color(Color::rgb(0.6, 0.6, 0.65))
                            )
                    )
            )
            .child(
                TextElement::new("Images show placeholder until texture loading is implemented")
                    .size(12.0)
                    .color(Color::rgb(0.5, 0.5, 0.55))
            )
            .into()
    }
}
